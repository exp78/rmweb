cmake_minimum_required(VERSION 3.16)
project(rmweb_auth_policy_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
add_executable(auth_policy_test ../auth_policy_test.cpp ../../engine/wpeqt/auth-policy.cpp)
target_include_directories(auth_policy_test PRIVATE ../../engine/wpeqt)
target_link_libraries(auth_policy_test PRIVATE Qt6::Core Qt6::Test)
target_compile_definitions(auth_policy_test PRIVATE QT_NO_KEYWORDS)
target_compile_options(auth_policy_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME auth_policy_test COMMAND auth_policy_test)
