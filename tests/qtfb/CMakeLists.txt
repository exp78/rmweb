cmake_minimum_required(VERSION 3.16)
project(rmweb_qtfb_tests LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "QTFB uses the AppLoad 64-bit Linux native socket ABI")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Test)
add_executable(qtfb_client_test ../qtfb_client_test.cpp
    ../../engine/wpeqt/qtfbclient.cpp ../../engine/wpeqt/qtfbclient.h)
target_link_libraries(qtfb_client_test PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
target_compile_definitions(qtfb_client_test PRIVATE QT_NO_KEYWORDS)
target_compile_options(qtfb_client_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME qtfb_client_test COMMAND qtfb_client_test)
set_tests_properties(qtfb_client_test PROPERTIES TIMEOUT 20)
