cmake_minimum_required(VERSION 3.16)
project(passkey_acceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WPE REQUIRED IMPORTED_TARGET wpe-webkit-2.0 wpe-platform-2.0 glib-2.0 gobject-2.0)
if(NOT TEST_ORIGIN MATCHES "^https://[a-z0-9]+([.-][a-z0-9]+)+(:[0-9]+)?$")
    message(FATAL_ERROR "Set TEST_ORIGIN to the dedicated exact HTTPS DNS origin")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/origin.h" "#ifndef ACCEPTANCE_ORIGIN\n#define ACCEPTANCE_ORIGIN \"${TEST_ORIGIN}\"\n#endif\n")
set(ENGINE "${CMAKE_CURRENT_SOURCE_DIR}/../../../engine/wpeqt")
set(SHARED "${ENGINE}/auth-policy.cpp" "${ENGINE}/auth-surface.cpp" "${ENGINE}/auth-passkey.cpp" "${ENGINE}/qtfbclient.cpp")
add_executable(acceptance-browser launcher.cpp ${SHARED})
target_include_directories(acceptance-browser PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(acceptance-browser PRIVATE Qt6::Core Qt6::Gui PkgConfig::WPE)
target_compile_definitions(acceptance-browser PRIVATE QT_NO_KEYWORDS)
target_compile_options(acceptance-browser PRIVATE -Wall -Wextra -Werror)
set_target_properties(acceptance-browser PROPERTIES
    INSTALL_RPATH "$ORIGIN/../runtime/lib"
    BUILD_WITH_INSTALL_RPATH TRUE)
add_executable(acceptance-policy-test policy-test.cpp)
target_link_libraries(acceptance-policy-test PRIVATE Qt6::Core)
add_executable(acceptance-entry "${ENGINE}/auth-entry.cpp")
set_target_properties(acceptance-entry PROPERTIES AUTOMOC OFF)
target_compile_options(acceptance-entry PRIVATE -Wall -Wextra -Werror -fstack-protector-strong)
target_link_options(acceptance-entry PRIVATE -Wl,-z,relro,-z,now)

add_executable(acceptance-engine-test engine-test.cpp ${SHARED})
target_include_directories(acceptance-engine-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(acceptance-engine-test PRIVATE Qt6::Core Qt6::Gui PkgConfig::WPE)
target_compile_definitions(acceptance-engine-test PRIVATE QT_NO_KEYWORDS ACCEPTANCE_ENGINE_FIXTURE=1 ACCEPTANCE_ORIGIN="https://login.example.com")
target_compile_options(acceptance-engine-test PRIVATE -Wall -Wextra -Werror)
